        exit(EXIT_FAILURE);
    }
}

static double real(void) {
    char *end;
    errno = 0;
    double value = strtod(input, &end);
    require(end != input && errno == 0);
    input = end;
    return value;
}

static void vd(int rank, const double *a, int n) {
    size_t size = 32 * ((size_t)n + 1);
    char *line = malloc(size);
    require(line != NULL);
    size_t used = (size_t)snprintf(line, size, "%d:", rank);
    for (int i = 0; i < n; i++)
        used += (size_t)snprintf(line + used, size - used, " %.12g", a[i]);
    puts(line);
    free(line);
}

int main(int argc, char **argv) {
    int r,k;
    MPI_Init(&argc,&argv);
    MPI_Comm_rank(W,&r);
    MPI_Comm_size(W,&k);
    require(argc==k+1);
    input=argv[r+1];

    require(k>=4 && k%2==0);
    int dims[2]={2,k/2},periods[2]={0,1},src,dst;
    MPI_Comm grid;
    MPI_Cart_create(W,2,dims,periods,0,&grid);
    MPI_Cart_shift(grid,1,1,&src,&dst);
    double a=real(),out;
    MPI_Sendrecv(&a,1,MPI_DOUBLE,dst,0,&out,1,MPI_DOUBLE,src,0,grid,MPI_STATUS_IGNORE);
    vd(r,&out,1); MPI_Comm_free(&grid);

    require(input[strspn(input," \t\r\n")]=='\0');
    MPI_Finalize();
    return 0;
}

#include <mpi.h>
#include <errno.h>
#include <limits.h>
#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define W MPI_COMM_WORLD

static const char *input;

static void require(int ok) {
    if (!ok) {
        int rank;
        MPI_Comm_rank(MPI_COMM_WORLD, &rank);
        fprintf(stderr, "%d-ранг. Кіріс деректері немесе процесс саны жарамсыз\n", rank);
        MPI_Abort(MPI_COMM_WORLD, 1);
        exit(EXIT_FAILURE);
    }
}

static int integer(void) {
    char *end;
    errno = 0;
    long value = strtol(input, &end, 10);
    require(end != input && errno == 0 && value >= INT_MIN && value <= INT_MAX);
    input = end;
    return (int)value;
}

static void vi(int rank, const int *a, int n) {
    size_t size = 32 * ((size_t)n + 1);
    char *line = malloc(size);
    require(line != NULL);
    size_t used = (size_t)snprintf(line, size, "%d:", rank);
    for (int i = 0; i < n; i++)
        used += (size_t)snprintf(line + used, size - used, " %d", a[i]);
    puts(line);
    free(line);
}

int main(int argc, char **argv) {
    int r,k;
    MPI_Init(&argc,&argv);
    MPI_Comm_rank(W,&r);
    MPI_Comm_size(W,&k);
    require(argc==k+1);
    input=argv[r+1];

    require(k==5 || k==7 || k==9);
    int index[k],edges[2*(k-1)],end=0;
    for(int i=0;i<k;i++) {
        if(i==0) {
            for(int j=1;j<k;j+=2) edges[end++]=j;
        } else if(i%2) {
            edges[end++]=0; edges[end++]=i+1;
        } else edges[end++]=i-1;
        index[i]=end;
    }
    MPI_Comm graph;
    MPI_Graph_create(W,k,index,edges,0,&graph);
    int count,a=integer();
    MPI_Graph_neighbors_count(graph,r,&count);
    int peers[count],out[count];
    MPI_Graph_neighbors(graph,r,count,peers);
    for(int i=1;i<count;i++) {
        int peer=peers[i],j=i;
        while(j>0 && peers[j-1]>peer) { peers[j]=peers[j-1]; j--; }
        peers[j]=peer;
    }
    for(int i=0;i<count;i++) {
        int p=peers[i];
        if(r<p) {
            MPI_Send(&a,1,MPI_INT,p,0,graph);
            MPI_Recv(out+i,1,MPI_INT,p,0,graph,MPI_STATUS_IGNORE);
        } else {
            MPI_Recv(out+i,1,MPI_INT,p,0,graph,MPI_STATUS_IGNORE);
            MPI_Send(&a,1,MPI_INT,p,0,graph);
        }
    }
    vi(r,out,count); MPI_Comm_free(&graph);

    require(input[strspn(input," \t\r\n")]=='\0');
    MPI_Finalize();
    return 0;
}
